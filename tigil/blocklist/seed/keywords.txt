# Tigil — heuristic rules (the anti-mirror engine)
#
# WHY THIS FILE IS THE MOST IMPORTANT ONE
#   PH gambling operators rotate domains constantly. "wpc2025 -> wpc2026 ->
#   wpc2027", "jiliko1.net -> jiliko2.net", a fresh .casino TLD every week.
#   A pure domain list loses that race by design. These rules match the *shape*
#   of the domain instead, so a mirror registered tomorrow is blocked today.
#
# HOW MATCHING WORKS
#   Rules are tested against the registrable domain only (e.g. for
#   "www.cdn.jiliko7.net" the tested string is "jiliko7.net"), lowercased.
#   The allowlist (allowlist.txt) is checked FIRST and always wins.
#
# SYNTAX
#   sub:<text>     substring match anywhere in the registrable domain
#   re:<regex>     Python/Java-compatible regex, searched (not anchored)
#   tld:<suffix>   the domain ends with this public suffix
#
# TUNING
#   Every rule here trades false positives for mirror coverage. Rules are
#   ordered loosest-risk first. If users report over-blocking, move the domain
#   into allowlist.txt rather than deleting a rule — the rule is earning its
#   keep elsewhere.

# --- The "JILI" ecosystem — the single most-promoted brand family in PH ----
# Deliberately broad: jili-anything in a domain is effectively always a casino
# skin. See allowlist.txt for the handful of legitimate collisions.
re:(^|[.-])jili
re:jili[0-9]{1,4}
re:(^|[.-])jl[0-9]{2,4}($|[.-])

# --- "bet" only when it looks like an operator brand, never bare -----------
# Blocks: 1xbet, 22bet, bet88, betso88, melbet, linebet, betwinner, 12bet.
# Does NOT block: diabetes, tibet, alphabet, betterhelp, bethesda.
re:^[0-9]{1,3}x?bet($|[0-9.-])
re:(^|[.-])bet[0-9]{2,4}($|[.-])
re:(^|[.-])bet(so|win|way|fair|king|pro|vip|club|game)[0-9]*($|[.-])
re:(win|mel|line|dafa|sports|crypto|cash)bet($|[0-9.-])

# --- "win" / "ph" operator brand shapes (22win, phlwin, ph365, mnl168) -----
re:(^|[.-])[0-9]{2,3}win($|[0-9.-])
re:(^|[.-])ph[a-z]{0,3}win($|[0-9.-])
re:(^|[.-])(ph|mnl|pinas|pinoy)[0-9]{3,4}($|[.-])
re:(^|[.-])wpc[0-9]{2,4}($|[.-])
re:(^|[.-])wpit[0-9]{2,3}($|[.-])
re:(^|[.-])sw[0-9]{3}($|[.-])
